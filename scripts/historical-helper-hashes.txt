# Every copy of each root helper script the toolkit ever shipped (issue #180):
# one `<path> <sha256>` line per distinct copy, for the scripts/ paths that
# historical-managed-paths.txt lists (the helpers early installers copied into
# a project's root scripts/ folder, byte for byte). Each sha256 is taken over the
# file with every carriage return removed, the way setup-project.js hashes the
# project's copy, so a CRLF checkout matches too. build-plugin.js emits these as
# managed-paths.json's historicalHelperHashes: a migration of a copy-install
# with no manifest removes a root helper only when its content matches a copy
# listed here, and keeps any other file of that name as the project's own.
#
# Committed data, so build-plugin.js --check gives the same answer without git
# history. scripts/test-build-plugin.js recomputes the list from the history and
# fails on any difference, printing the lines it expects. The copies stopped
# changing when the helpers moved under .claude/scripts/ (v4.3.0); regenerate
# only when historical-managed-paths.txt gains a root helper. From the
# repository root:
#   for p in $(grep '^scripts/' scripts/historical-managed-paths.txt); do
#     for c in $(git log --full-history --format=%H HEAD -- "$p"); do
#       git cat-file -e "$c:$p" 2>/dev/null &&
#         echo "$p $(git show "$c:$p" | tr -d '\r' | sha256sum | cut -d' ' -f1)"
#     done
#   done | sort -u
